            \item Se $\textcolor{blue}{y}$ è la prima lettera maggiore di $\textcolor{red}{x}$:

            \item \emph{Inserimento speciale} se $\textcolor{red}{x}$ compariva

\begin{frame}{Corrispondenza di Edelman--Greene}
    \[
        \Biggl\{
            \text{
                \begin{tabular}{c}
                    Parole ridotte\\
                    della permutazione\\
                    $w \in S_n$
                \end{tabular}
            }
        \Biggr\}
        \; \longleftrightarrow \;
        \Biggl\{
            (P, Q)
            \Biggm\vert
            \text{
                \begin{tabular}{c}
                    $P$ e $Q$ stessa forma,\\
                    $P$ è \emph{ridotto} per $w$,\\
                    $Q$ è SYT
                \end{tabular}
            }
        \Biggr\}
    \]

    \begin{columns}
        \column{.4\textwidth}
        \begin{center}
            \ytableaunumber{1 2 3, 2 3, 3}
        \end{center}
        \column{.6\textwidth}
        C'è solo \emph{un} tableau ridotto per l'elemento più 
        lungo in $S_n$.
    \end{columns}

\end{frame}
