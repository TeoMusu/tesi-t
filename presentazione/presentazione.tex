\begin{frame}
    \[
        \text{Parola ridotta: $132312$}
    \]

    \[
        \ytableausetup{aligntableaux=top}
        \ytableaunumber{1}
        \xrightarrow{\leftarrow 3}
        \ytableaunumber{1 3}
        \xrightarrow{\leftarrow 2}
        \ytableaunumber{1 2, 3}
        \xrightarrow{\leftarrow 3}
        \ytableaunumber{1 2 3, 3}
        \xrightarrow{\leftarrow 1}
        \ytableaunumber{1 2 3, 2, 3}
        \xrightarrow{\leftarrow 2}
        \ytableaunumber{1 2 3, 2 3, 3}
        % = P(v).
    \]

    Inserimento della lettera $x$:
    \begin{itemize}
        \item se $x$ è più grande di tutta la riga, aggiungi $x$ in fondo alla riga;
        \item altrimenti se $y$ è la prima lettera maggiore di $x$,
            $y$ viene \emph{spinta} nella riga successiva e $x$ prende il suo posto,
            \emph{a meno che\dots}
        \item \emph{eccezione:} se $x$ compariva già nella riga e $y = x+1$,
            allora viene spinto $y$, ma la riga rimane \emph{invariata}.
    \end{itemize}


\end{frame}

\begin{frame}{$P$-tableau}
    \begin{columns}
        \column{.4\textwidth}
        \begin{center}
            \ytableaunumber{1 2 3, 2 3, 3}
        \end{center}

        \column{.6\textwidth}
        Il $P$-tableau della parola ridotta $132312$.
        \begin{itemize}
            \item Righe e colonne crescenti.
            \item Il suo \emph{row-reading}
                \[
                    3 \, 23 \, 123
                \]
                è una parola ridotta per la stessa permutazione.
            \item Un tableau di questo tipo si dice \emph{ridotto}.
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Corrispondenza di Edelman--Greene}
    \[
        \Biggl\{
            \text{
                \begin{tabular}{c}
                    Parole ridotte\\
                    della permutazione\\
                    $w \in S_n$
                \end{tabular}
            }
        \Biggr\}
        \; \longleftrightarrow \;
        \Biggl\{
            (P, Q)
            \Biggm\vert
            \text{
                \begin{tabular}{c}
                    $P$ e $Q$ stessa forma,\\
                    $P$ è \emph{ridotto} per $w$,\\
                    $Q$ è SYT
                \end{tabular}
            }
        \Biggr\}
    \]

    \begin{columns}
        \column{.4\textwidth}
        \begin{center}
            \ytableaunumber{1 2 3, 2 3, 3}
        \end{center}
        \column{.6\textwidth}
        C'è solo \emph{un} tableau ridotto per l'elemento più 
        lungo in $S_n$.
    \end{columns}

\end{frame}
